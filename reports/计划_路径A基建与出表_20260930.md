```python
# -*- coding: utf-8 -*-
"""产物目录 manifest 指纹(设计 reports/调研_阻塞与开发界方案_20260930.md §4.2)。

make:  逐文件 md5 → 目录下 _manifest.json(generated_by/code_commit/files)。
check: 现有文件对照 manifest → 新增/缺失/变更;有漂移退出码 1。
globs: 只收匹配的相对路径(fnmatch),缺省全部文件;_manifest.json 自身永不入账。

用法:
  python src/tools/make_manifest.py make  <目录> [--glob 模式]...
  python src/tools/make_manifest.py check <目录> [--glob 模式]...
"""
from __future__ import annotations

import argparse
import fnmatch
import hashlib
import json
import subprocess
import sys
from pathlib import Path

NAME = "_manifest.json"
_ROOT = Path(__file__).resolve().parents[2]


def _md5(p: Path) -> str:
    h = hashlib.md5()
    with open(p, "rb") as f:
        for chunk in iter(lambda: f.read(1 << 20), b""):
            h.update(chunk)
    return h.hexdigest()


def _code_commit() -> str:
    try:
        return subprocess.run(["git", "rev-parse", "--short", "HEAD"], cwd=_ROOT,
                              capture_output=True, text=True, check=True).stdout.strip()
    except Exception:                                # noqa: BLE001
        return "unknown"


def _files(root: Path, globs: tuple[str, ...]) -> dict[str, str]:
    out = {}
    for p in sorted(root.rglob("*")):
        if not p.is_file() or p.name == NAME:
            continue
        rel = p.relative_to(root).as_posix()
        if globs and not any(fnmatch.fnmatch(rel, g) for g in globs):
            continue
        out[rel] = _md5(p)
    return out


def make(root: str | Path, globs: tuple[str, ...] = ()) -> Path:
    root = Path(root)
    out = {"generated_by": "src/tools/make_manifest.py", "code_commit": _code_commit(),
           "files": _files(root, globs)}
    root.mkdir(parents=True, exist_ok=True)
    (root / NAME).write_text(json.dumps(out, ensure_ascii=False, indent=1), encoding="utf-8")
    return root / NAME


def check(root: str | Path, globs: tuple[str, ...] = ()) -> dict:
    root = Path(root)
    old = json.load(open(root / NAME, encoding="utf-8"))["files"]
    now = _files(root, globs)
    return {"added": sorted(set(now) - set(old)),
            "missing": sorted(set(old) - set(now)),
            "changed": sorted(k for k in set(now) & set(old) if now[k] != old[k])}


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("cmd", choices=("make", "check"))
    ap.add_argument("root")
    ap.add_argument("--glob", action="append", default=[])
    a = ap.parse_args()
    if a.cmd == "make":
        print(f"manifest → {make(a.root, tuple(a.glob))}")
        return
    r = check(a.root, tuple(a.glob))
    print(json.dumps(r, ensure_ascii=False, indent=1))
    sys.exit(1 if (r["added"] or r["missing"] or r["changed"]) else 0)


if __name__ == "__main__":
    main()
```

```python
# -*- coding: utf-8 -*-
"""数据级 diff 核验器(设计 reports/调研_阻塞与开发界方案_20260930.md §4.1)。

行线:按 line_id 对齐逐点比残差——两代数据集行判定源 md5 相同,几何应逐位相等,
     残差 >TOL 即真变更(新增/消失/漂移/点数不同),不做像素 diff、不猜原因。
列线/框边:两代表征不同(v1 整线 / v2 逐碎片段),只出计数与 backing 账,不做点级对齐。
变更行线出 1:1 对照裁片(旧|新并排),底图用两代各自的叠加图;缺叠加图只记账不裁片。
退出码:行线有任一变更 → 1(两代同源时这就是"行几何被谁动了"的报警)。

用法:
  python -u -X utf8 src/tools/review_diff.py --old <v1目录> --new <v2目录> \
      [--stems 1015 1019 ...] [--out runs/tmp/review_diff_20260930]
"""
from __future__ import annotations

import argparse
import json
import sys
from pathlib import Path

import numpy as np
from PIL import Image

Image.MAX_IMAGE_PIXELS = None

ROOT = Path(__file__).resolve().parents[2]
if str(ROOT) not in sys.path:
    sys.path.insert(0, str(ROOT))

TOL = 0.5          # px;同源行线残差实测应逐位为 0,0.5 只吸收浮点落盘误差
PAD = 150          # 裁片上下文外扩


def _load(ds: Path, stem: str) -> dict:
    return json.load(open(ds / "每张" / f"钻孔柱状图{stem}.json", encoding="utf-8"))


def diff_rows(old: dict, new: dict, tol: float = TOL) -> dict:
    a = {l["line_id"]: np.asarray(l["points"], float)
         for l in old["lines"] if l["kind"] == "row"}
    b = {l["line_id"]: np.asarray(l["points"], float)
         for l in new["lines"] if l["kind"] == "row"}
    moved, res = [], []
    for k in sorted(set(a) & set(b)):
        if a[k].shape != b[k].shape:
            moved.append({"line_id": k, "why": "点数不同", "d": None})
            continue
        d = float(np.abs(a[k] - b[k]).max())
        res.append(d)
        if d > tol:
            moved.append({"line_id": k, "why": "坐标漂移", "d": round(d, 3)})
    r = np.asarray(res, float)
    return {"n_old": len(a), "n_new": len(b), "n_moved": len(moved), "moved": moved,
            "added": sorted(set(b) - set(a)), "removed": sorted(set(a) - set(b)),
            "residual": {"max": float(r.max()) if r.size else None,
                         "p99": float(np.percentile(r, 99)) if r.size else None}}


def diff_cols(old: dict, new: dict) -> dict:
    def acc(d):
        col = [l for l in d["lines"] if l["kind"] == "col"]
        edge = [l for l in d["lines"] if l["kind"] == "edge"]
        return {"col_lines": len({l["col_index"] for l in col}),
                "col_segs": len(col),
                "edge_segs": len(edge),
                "backing": {k: sum(1 for l in col if l.get("backing") == k)
                            for k in ("frag", "fill", "extend")}}
    return {"old": acc(old), "new": acc(new)}


def _crops(stem: str, moved: list, old: dict, new: dict,
           old_dir: Path, new_dir: Path, out_dir: Path) -> list:
    """变更行线的 旧|新 1:1 对照裁片;缺图或缺线返回跳过名单。"""
    skips, made = [], []
    ov_old = old_dir / "叠加图" / f"钻孔柱状图{stem}.png"
    ov_new = new_dir / "叠加图" / f"钻孔柱状图{stem}.png"
    if not (ov_old.exists() and ov_new.exists()):
        return [], ["缺叠加图,全部跳过"]
    pts = {l["line_id"]: np.asarray(l["points"], float)
           for d in (old, new) for l in d["lines"] if l["kind"] == "row"}
    im_old = Image.open(ov_old)
    im_new = Image.open(ov_new)
    (out_dir / "局部1比1").mkdir(parents=True, exist_ok=True)
    for m in moved:
        k = m["line_id"]
        if k not in pts:
            skips.append(f"{k} 点集缺失")
            continue
        p = pts[k]
        y0 = max(0, int(p[:, 1].min()) - PAD)
        y1 = min(im_new.height, int(p[:, 1].max()) + PAD)
        x0 = max(0, int(p[:, 0].min()) - PAD)
        x1 = min(im_new.width, int(p[:, 0].max()) + PAD)
        box = (x0, y0, x1, y1)
        a, b = im_old.crop(box), im_new.crop(box)
        w, h = a.width + b.width + 4, max(a.height, b.height)
        canvas = Image.new("RGB", (w, h), (255, 255, 255))
        canvas.paste(a, (0, 0))
        canvas.paste(b, (a.width + 4, 0))
        fp = out_dir / "局部1比1" / f"{stem}_{k}.png"
        canvas.save(fp)
        made.append(str(fp))
    return made, skips


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--old", required=True)
    ap.add_argument("--new", required=True)
    ap.add_argument("--stems", nargs="*", default=[])
    ap.add_argument("--out", default="runs/tmp/review_diff_20260930")
    a = ap.parse_args()
    old_dir, new_dir, out_dir = Path(a.old), Path(a.new), Path(a.out)
    stems = a.stems or sorted(p.stem.removeprefix("钻孔柱状图")
                              for p in (new_dir / "每张").glob("*.json"))
    report = {"old": str(old_dir), "new": str(new_dir), "stems": {}}
    alarm = 0
    for stem in stems:
        old, new = _load(old_dir, stem), _load(new_dir, stem)
        rows, cols = diff_rows(old, new), diff_cols(old, new)
        crops, skips = ([], [])
        if rows["n_moved"] or rows["added"] or rows["removed"]:
            crops, skips = _crops(stem, rows["moved"], old, new, old_dir, new_dir, out_dir)
        n_change = rows["n_moved"] + len(rows["added"]) + len(rows["removed"])
        alarm += n_change
        report["stems"][stem] = {"rows": rows, "cols": cols,
                                 "crops": crops, "skips": skips}
        print(f"{stem}: 行线 变更 {n_change}"
              f"(漂移 {rows['n_moved']} / 新增 {len(rows['added'])} / 消失 {len(rows['removed'])})"
              f" 残差max {rows['residual']['max']};列账 old{cols['old']['col_segs']}"
              f"→new{cols['new']['col_segs']}段")
    out_dir.mkdir(parents=True, exist_ok=True)
    fp = out_dir / "diff账目.json"
    fp.write_text(json.dumps(report, ensure_ascii=False, indent=1), encoding="utf-8")
    print(f"账目 → {fp};行线变更合计 {alarm}")
    sys.exit(1 if alarm else 0)


if __name__ == "__main__":
    main()
```

```python
def _sep_bounds(stem: str) -> list:
    """列/框边界真段几何(单一真源 = col_segments.line_parts,即 _分割线.json 的 P10 口径)。

    parts=真段(无账弦),bridged=有账补全(fill/extend);dropped(带内图案碎片)不入选。
    返回边界列表(原序),每项 {kind,index,side,ys,xs},ys 升序,供 np.interp 取 x。
    真断口处 interp 线性跨接——只作切格裁片的导引,不作为结构线外传(与 6a2 fill 同性质)。
    """
    lp = col_segments.line_parts(stem)
    bounds = []
    for ln in lp["lines"]:
        pts = [p for seg, _ in ln["parts"] for p in seg]
        pts += [p for seg, _ in ln.get("bridged", []) for p in seg]
        if len(pts) < 2:
            continue
        pts = sorted(pts, key=lambda p: p[1])          # [[x,y],...] → y 升序
        bounds.append({"kind": ln["kind"], "index": ln.get("index"),
                       "side": ln.get("side"),
                       "xs": [float(p[0]) for p in pts], "ys": [float(p[1]) for p in pts]})
    return bounds
```
